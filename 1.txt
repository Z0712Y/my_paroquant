===========================（4）实验评估===========================

==========1.非推理任务（假量化，BoolQ、ARC-C、ARC-E、HSwag 等）==========
#设置离线模式，使用本地数据集
export HF_DATASETS_OFFLINE=1 && export TRANSFORMERS_OFFLINE=1
./experiments/tasks/non_reasoning.sh models/Qwen3-14B-PARO-pseudo

==========2.推理任务（假量化，MMLU-Pro、GPQA、AIME 等）==========
# 在新的虚拟环境执行推理任务，设置离线模式，使用本地数据集
# conda env create -f ./experiments/tasks/reasoning/environment.yml
conda activate paroquant-eval
export HF_DATASETS_OFFLINE=1 && export TRANSFORMERS_OFFLINE=1

# MMLU-PRO：只跑种子 42
./experiments/tasks/reasoning.sh models/Qwen3-14B-PARO-pseudo 42 MMLU-PRO

# 其他三个任务：跑种子 42, 0, 1，然后取平均
for seed in 42 0 1; do
    ./experiments/tasks/reasoning.sh models/Qwen3-14B-PARO-pseudo $seed GPQA-Diamond
    ./experiments/tasks/reasoning.sh models/Qwen3-14B-PARO-pseudo $seed AIME-2024
    ./experiments/tasks/reasoning.sh models/Qwen3-14B-PARO-pseudo $seed AIME-2025
done

==========3.吞吐量测试（真量化）==========
./experiments/throughput/bench.sh models/Qwen3-14B-PARO
