===========================（3）生成可部署模型===========================
# 伪量化用于验证精度，存储占用与FP16一致，pseudo quantization
python3 scripts/pseudo_quant.py \
    --model /home/hhw/zy/models/Qwen3-8B \
    --result-dir output/Qwen3-8B \
    --output-path models/Qwen3-8B-GUIDED-pseudo

# 真实量化，存储占用减少3/4，用于实际部署和推理，real quantization
python3 scripts/real_quant.py \
    --model /home/hhw/zy/models/Qwen3-8B \
    --result-dir output/Qwen3-8B \
    --output-path models/Qwen3-8B-GUIDED \
    --group-size 128 \
    --n-bit 4
